```sh
python3 - <<'PY'
import hashlib, json, pathlib, re
root = pathlib.Path.cwd()
files = [root / "README.md"]
for folder in ("docs/implementation", "docs/adr", "tests/reports"):
    files.extend(sorted((root / folder).glob("*.md")))
links = 0
for path in files:
    content = path.read_text()
    assert content.strip(), path
    if path != root / "README.md":
        assert content.isascii(), path
    for target in re.findall(r"\[[^\]\n]+\]\(([^)\n]+)\)", content):
        if target.startswith(("https://", "http://", "mailto:", "#")):
            continue
        assert (path.parent / target.split("#", 1)[0]).exists(), (path, target)
        links += 1
json_files = list(root.rglob("*.json"))
for path in json_files:
    json.loads(path.read_text())
manifest = json.loads((root / "MANIFEST.json").read_text())
changed = [entry["path"] for entry in manifest["files"]
           if hashlib.sha256((root / entry["path"]).read_bytes()).hexdigest()
           != entry["sha256"]]
assert changed == ["README.md"], changed
print({"markdown": len(files), "local_links": links,
       "json": len(json_files), "original_changes": changed})
PY
```
